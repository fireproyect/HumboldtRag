async function subirArchivo() { const fileInput = document.getElementById("file-input"); const uploadBtn = document.getElementById("upload-btn"); const statusDiv = document.getElementById("status-message"); if (fileInput.files.length === 0) { statusDiv.style.color = "#ef4444"; statusDiv.innerText = "Por favor selecciona un archivo primero."; return; } const archivo = fileInput.files[0]; const formData = new FormData(); formData.append("file", archivo); uploadBtn.disabled = true; statusDiv.style.color = "var(--brand-accent)"; statusDiv.innerText = `Subiendo "${archivo.name}" al servidor...`; try { const response = await fetch(`${API_URL}/admin/subir`, { method: "POST", headers: getAuthHeaders(), body: formData }); const resultado = await response.json(); if (!response.ok) throw new Error(resultado.detail || "Error al subir el archivo."); statusDiv.innerText = resultado.mensaje; fileInput.value = ""; if (resultado.task_id) { await monitorearTarea(resultado.task_id, statusDiv, uploadBtn); } else { uploadBtn.disabled = false; cargarDashboard(); } } catch (error) { console.error(error); statusDiv.style.color = "#ef4444"; statusDiv.innerText = `Error: ${error.message}`; uploadBtn.disabled = false; } } async function monitorearTarea(taskId, statusDiv, uploadBtn) { const intervalo = setInterval(async () => { try { const res = await fetch(`${API_URL}/admin/tarea/${taskId}`, { headers: getAuthHeaders() }); if (!res.ok) return; const estado = await res.json(); statusDiv.innerText = estado.mensaje; if (estado.status === "completado") { clearInterval(intervalo); statusDiv.style.color = "#16a34a"; uploadBtn.disabled = false; cargarDashboard(); } else if (estado.status === "error") { clearInterval(intervalo); statusDiv.style.color = "#ef4444"; uploadBtn.disabled = false; } } catch (err) { console.warn("Reintentando consulta de estado...", err); } }, 4000); }
